89
90 len = std::min (len, free());
91 ::memcpy (_in, buf, len);
92 _in += len;

99 return 0;
100 }
101 return -1;

104 void seek (uint16_t len) {
105
106 len = std::min (len, static_cast<uint16_t>(_size - static_cast<uint16_t>(_out - _data)));
107 _in += len;
108 }
109
